            const d = new Date();

            const month = String(d.getMonth()+1).padStart(2,'0');
            const day = String(d.getDate()).padStart(2,'0');
            return `${year}-${month}-${day}`;

            const year = date.getFullYear();
            const month = String(date.getMonth()+1).padStart(2,'0');
            const day = String(date.getDate()).padStart(2,'0');
            return `${year}-${month}-${day}`;

        function showToast(msg, type='success'){ 
            const container = document.getElementById('toast-container');
            const el = document.createElement('div');
            
            let bgColor, iconClass, borderClass;
            
            if (type === 'success') {
                bgColor = 'bg-harmony-accent'; 
                iconClass = 'fa-check-circle';
            } else if (type === 'error') {
                bgColor = 'bg-harmony-error'; 
                iconClass = 'fa-times-circle';
            } else { // warning
                bgColor = 'bg-harmony-warning'; 
                iconClass = 'fa-exclamation-triangle';
            }

            el.className = `${bgColor} text-white px-6 py-4 rounded-xl shadow-2xl flex items-center gap-3 min-w-[300px] transform transition-all pointer-events-auto toast-enter border-l-4 border-white`;

            container.appendChild(el);
            setTimeout(() => {
                el.classList.remove('toast-enter'); 
                el.classList.add('toast-exit');
                setTimeout(() => el.remove(), 500); 
            }, 3500);
        }

        // Fungsi untuk merender Kartu Antrian Selanjutnya di Dashboard
        function renderNextQueueCard(antrianList) {
            const container = document.getElementById('next-queue-content');
            
            if (!antrianList || antrianList.length === 0) {
                container.innerHTML = '<div class="text-center py-4 text-slate-400 italic">Tidak ada antrian aktif/booking terdekat.</div>';
                return;
            }

            const nextQueue = antrianList[0];

            const today = getLocalDate();

            const isBooking = nextQueue.tgl_kunjungan > today;
            let statusText = isBooking ? 'Booking' : nextQueue.status;
            let statusColor = 'bg-harmony-primary_light text-harmony-primary';
            let queueBg = 'bg-white';

            if (isBooking) {
                 statusColor = 'bg-harmony-accent_light text-harmony-accent';
                 queueBg = 'bg-harmony-accent_light/50';
            } else if (nextQueue.status === 'Menunggu Obat') {
                statusColor = 'bg-harmony-warning_light text-harmony-warning';
            } else if (nextQueue.status === 'Sedang Pemeriksaan') {
                statusColor = 'bg-harmony-secondary_light text-harmony-secondary';
            } else if (nextQueue.status === 'Menunggu Pemeriksaan' && nextQueue.tgl_kunjungan <= today) {
                 statusColor = 'bg-harmony-primary_light text-harmony-primary';
                 queueBg = 'bg-harmony-primary_light/50';
            }


            container.innerHTML = `
                <div class="flex flex-col md:flex-row items-center gap-6 p-4 rounded-xl border border-slate-200 ${queueBg} hover:shadow-md transition duration-300">
                    <!-- No. Antrian -->
                    <div class="p-4 rounded-xl w-32 flex-shrink-0 text-center bg-white border border-slate-200 shadow-sm">
                        <span class="text-xs uppercase tracking-widest text-slate-500 font-bold">${isBooking ? 'Tgl Kunjung' : 'Antrian'}</span>
                        <span class="text-4xl font-extrabold text-slate-800 block">${isBooking ? nextQueue.tgl_kunjungan.slice(-2) : nextQueue.no_antrian}</span>
                    </div>

                    <!-- Detail & Status -->
                    <div class="flex-1 space-y-1">
                        <div class="flex justify-between items-start">
                            <h4 class="text-xl font-extrabold text-slate-800">${nextQueue.nama_poli}</h4>
                            <span class="px-3 py-1 rounded-full text-xs font-bold ${statusColor}">${statusText}</span>
                        </div>
                        <p class="text-sm text-slate-600 flex items-center gap-2">
                            <i class="fas fa-user-md text-harmony-secondary"></i> ${nextQueue.nama_dokter} <span class="text-xs text-slate-400">(${nextQueue.spesialis || '-'})</span>
                        </p>
                        <div class="text-sm text-slate-500 pt-1 border-t border-slate-100 mt-2 flex justify-between">
                            <span class="flex items-center gap-1 font-semibold"><i class="far fa-calendar"></i> ${nextQueue.tgl_kunjungan}</span>

        function toggleDaftarTab(tab, forceLoad = true) {
            const tabForm = document.getElementById('tab-form');
            const tabList = document.getElementById('tab-list');
            const sectionForm = document.getElementById('daftar-form-section');
            const sectionList = document.getElementById('daftar-list-section');

            tabForm.classList.add('border', 'border-slate-300');

                sectionForm.classList.remove('hidden');
                sectionList.classList.add('hidden');

                tabForm.classList.remove('border', 'border-slate-300');
                tabForm.classList.add('bg-white', 'text-slate-600');

        function loadAllQueues() {
            if(!currentUser) return;
            const container = document.getElementById('all-queues-container');
            const dateVal = document.getElementById('queue-filter-date').value;
            const statusVal = document.getElementById('queue-filter-status').value;
            
            container.innerHTML = '<p class="text-slate-400 text-sm italic text-center py-4"><i class="fas fa-spinner fa-spin mr-2"></i> Memuat antrian...</p>';
            
            let finalStatus = statusVal;
            let finalDate = dateVal;

            if (statusVal === 'Booking') {
                finalStatus = 'Menunggu Pemeriksaan'; 
                finalDate = ''; 
            }

            fetch(`${API_URL}?action=get_all_active_queues&nik=${currentUser.nik}&date=${finalDate}&status=${finalStatus}`)
            .then(r => r.json())
            .then(data => {
                if (data.length === 0) {
                    container.innerHTML = '<p class="text-slate-500 text-sm italic text-center py-6">Tidak ada antrian ditemukan untuk filter ini.</p>';
                    return;
                }

                const todayISO = getLocalDate();
                
                if (statusVal === 'Booking') {
                    data = data.filter(queue => queue.tgl_kunjungan > todayISO);
                }

                let html = '';
                data.forEach(queue => {
                    
                    const isBookingFuture = queue.tgl_kunjungan > todayISO;
                    
                    let statusColor = 'bg-harmony-primary_light text-harmony-primary';
                    let statusText = queue.status;
                    let queueBg = 'bg-white';
                    
                    if (isBookingFuture) {
                         statusText = 'Booking';
                         statusColor = 'bg-harmony-accent_light text-harmony-accent';
                         queueBg = 'bg-harmony-accent_light/50';
                    } else if (queue.status === 'Menunggu Obat') {
                        statusColor = 'bg-harmony-warning_light text-harmony-warning';
                    } else if (queue.status === 'Sedang Pemeriksaan') {
                        statusColor = 'bg-harmony-secondary_light text-harmony-secondary';
                    } else if (queue.status === 'Menunggu Pemeriksaan' && queue.tgl_kunjungan <= todayISO) {
                         statusColor = 'bg-harmony-primary_light text-harmony-primary';
                         queueBg = 'bg-harmony-primary_light/50';
                    }
                    
                    // MENGHAPUS FUNGSIONALITAS PEMBATALAN ANTRIAN (Sesuai Permintaan)
                    const cancelBtn = `<span class="text-slate-500 px-2 py-1 text-xs font-medium"></span>`;


                    html += `
                        <div class="flex items-start gap-4 p-4 rounded-xl border border-slate-200 ${queueBg} hover:shadow-md transition interactive-card">
                            <!-- Antrian Number / Tgl -->
                            <div class="p-3 rounded-lg flex-shrink-0 text-center bg-white border border-slate-200 w-24 shadow-sm">
                                <span class="text-xs uppercase tracking-widest text-slate-500 font-bold">${isBookingFuture ? 'Tgl' : 'Antrian'}</span>
                                <span class="text-2xl font-extrabold text-slate-800 block">${isBookingFuture ? queue.tgl_kunjungan.slice(-2) : queue.no_antrian}</span>
                            </div>

                            <!-- Detail -->
                            <div class="flex-1 space-y-1">
                                <div class="flex justify-between items-start">
                                    <h4 class="text-lg font-bold text-slate-800">${queue.nama_poli}</h4>
                                    <span class="px-3 py-1 rounded-full text-xs font-bold ${statusColor}">${statusText}</span>
                                </div>
                                <p class="text-sm text-slate-600 flex items-center gap-2">
                                    <i class="fas fa-user-md text-harmony-secondary"></i> ${queue.nama_dokter}
                                </p>
                                <div class="text-xs text-slate-500 pt-1 flex gap-4">
                                    <span class="flex items-center gap-1"><i class="far fa-calendar"></i> ${queue.tgl_kunjungan}</span>
                                    <span class="flex items-center gap-1"><i class="far fa-clock"></i> ${queue.jam_kunjungan || '-'} WIB</span>
                                </div>
                            </div>
                            
                            <!-- Aksi (Diganti dengan pesan "Pembatalan Tidak Diizinkan") -->
                            <div class="self-start">
                                ${cancelBtn}
                            </div>
                        </div>
                    `;
                });
                container.innerHTML = html;
            }).catch(() => {
                container.innerHTML = '<p class="text-slate-500 text-sm italic text-center py-6 text-harmony-error">Gagal memuat antrian (Koneksi Error).</p>';
            });
        }

        /**
         * Mengambil dan menampilkan detail struk pembayaran untuk Pasien.
         * Pasien hanya dapat melihat detail jika transaksi sudah Lunas.
         * @param {number} id_pendaftaran ID Pendaftaran yang sudah lunas.
         */
        window.viewStrukPasien = async function(id_pendaftaran) {
            showToast('Memuat detail struk...', 'info');
            openModal('modal-struk-pasien'); // Tampilkan modal dengan loading
            
            try {
                // Panggil endpoint baru di backend_pasien.php
                const res = await fetch(`${API_URL}?action=get_struk_detail_pasien&id=${id_pendaftaran}`);
                const d = await res.json();
                
                if (d.status === 'success' && d.data) {
                    const data = d.data;
                    
                    // 1. Set Info Umum
                    document.getElementById('struk-pasien-tgl-lunas').textContent = new Date(data.tgl_selesai).toLocaleString('id-ID', { day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' });
                    document.getElementById('struk-pasien-id-pendaftaran').textContent = data.id;
                    document.getElementById('struk-pasien-nama-pasien').textContent = data.nama_pasien;
                    document.getElementById('struk-pasien-no-rm').textContent = data.no_rm;
                    document.getElementById('struk-pasien-layanan').textContent = `${data.nama_poli} (${data.nama_dokter})`;

                    // 2. Set Rincian Biaya
                    document.getElementById('struk-pasien-biaya-medis').textContent = formatRupiah(data.biaya_medis_dasar);
                    document.getElementById('struk-pasien-biaya-obat').textContent = formatRupiah(data.total_harga_resep);

                    // 3. Set Detail Resep
                    const resepContainer = document.getElementById('struk-pasien-detail-resep');
                    resepContainer.innerHTML = '';
                    
                    if (data.resep_detail && data.resep_detail.length > 0) {
                        data.resep_detail.forEach(resep => {
                            const itemEl = document.createElement('p');
                            const satuan = resep.satuan || 'Unit';
                            itemEl.textContent = `- ${resep.nama} (${resep.jumlah} ${satuan})`; 
                            resepContainer.appendChild(itemEl);
                        });
                        document.getElementById('struk-pasien-detail-resep-container').classList.remove('hidden');
                    } else {
                        const itemEl = document.createElement('p');
                        itemEl.textContent = `- Tidak ada resep obat`;
                        resepContainer.appendChild(itemEl);
                    }

                    // 4. Set Total dan Pembayaran
                    document.getElementById('struk-pasien-total-tagihan').textContent = formatRupiah(data.total_biaya_final);
                    // Gunakan data dari server untuk jumlah bayar dan kembalian
                    document.getElementById('struk-pasien-jumlah-bayar').textContent = formatRupiah(data.jumlah_bayar_input); 
                    document.getElementById('struk-pasien-kembalian').textContent = formatRupiah(data.kembalian); 
                    
                } else {
                    showToast(d.message || 'Gagal memuat data struk. Pastikan tagihan sudah lunas.', 'error');
                    closeModal('modal-struk-pasien');
                }

            } catch (e) {
                console.error("Error viewing struk:", e);
                showToast('Gagal terhubung ke server (Error: Network/Server Pasien).', 'error');
                closeModal('modal-struk-pasien');
            }
        }

            openModal('modal-rm-pasien');
            fetch(`${API_URL}?action=get_rm_detail_pasien&id_rm=${id_rm}&nik=${currentUser.nik}`)
            .then(r => r.json())
            .then(resp => {

                    document.getElementById('modal-rm-dokter').textContent = d.nama_dokter;
                    document.getElementById('modal-rm-poli').textContent = d.nama_poli;

                    document.getElementById('modal-rm-diagnosa-utama').textContent = d.diagnosa_utama_display || '-'; 
                    document.getElementById('modal-rm-diagnosa-sekunder').textContent = d.diagnosa_sekunder_display || '-'; 
                    document.getElementById('modal-rm-instruksi').textContent = d.instruksi_pasien || '-';

                    let resepHTML = '';
                    try {
                        const resepData = d.resep_obat ? JSON.parse(d.resep_obat) : [];
                        const obatArr = Array.isArray(resepData) ? resepData : [];

                        if(obatArr.length > 0) {
                            obatArr.forEach(o => {
                                resepHTML += `
                                <div class="bg-white p-3 rounded-xl border border-slate-200 shadow-sm">
                                    <p class="font-bold text-slate-800 text-sm">${o.nama} <span class="text-xs text-harmony-primary">(${o.dosis || '-'})</span></p>
                                    <div class="flex justify-between items-center mt-1 text-xs text-slate-500">

                        console.error("Error parsing resep obat:", e);

                    }
                    
                    document.getElementById('modal-rm-resep-list').innerHTML = resepHTML;

                    showToast(resp.message, 'error');
                }
            });
        }

        window.onDokterSelected = function() {
            const id = document.getElementById('dokter-select').value; 
            const doc = dokterDataGlobal.find(x=>x.id==id);
            
            document.getElementById('step-tanggal').classList.add('opacity-50');
            document.getElementById('day-buttons-container').innerHTML = '<div class="w-full text-center py-6 text-slate-400 italic bg-slate-50 rounded-xl border border-dashed border-slate-200">Silahkan pilih dokter terlebih dahulu</div>';
            document.getElementById('result-info').classList.add('hidden');
            document.getElementById('btn-daftar').disabled = true;
            document.getElementById('tgl-kunjungan-input').value = '';

            if (!doc) return;

            document.getElementById('step-tanggal').classList.remove('opacity-50');
            
            document.getElementById('day-buttons-container').innerHTML = '<div class="w-full text-center py-4"><i class="fas fa-spinner fa-spin text-harmony-primary text-2xl"></i></div>';

            fetch(API_URL+'?action=get_quota_info&id_dokter='+id).then(r=>r.json()).then(d=>{
                let h=''; const max=parseInt(d.max_quota); const schedule = parseSchedule(doc.jadwal_praktik);
                const dates = genDates(schedule);
                
                if(dates.length === 0) h = '<div class="w-full text-center text-harmony-error py-4 bg-harmony-error_light rounded-xl">Jadwal Penuh / Tidak Tersedia</div>';
                else {
                    dates.forEach(dt=>{
                        const booked = d.booked[dt.val] ? parseInt(d.booked[dt.val]) : 0;
                        const sisa = max - booked;
                        const isFull = sisa <= 0;
                        
                        h+=`
                        <button type="button" class="border transition-all p-4 rounded-xl text-center w-32 flex-shrink-0 flex flex-col items-center justify-center gap-1 interactive-card
                            ${isFull ? 'bg-slate-50 text-slate-400 cursor-not-allowed' : 'bg-white border-slate-300 hover:border-harmony-primary day-btn'}" 
                            ${isFull?'disabled':'onclick="selDate(this, \''+dt.val+'\', \''+schedule[dt.day].s+'\', \''+schedule[dt.day].e+'\')"'}>
                            <span class="text-xs font-bold uppercase">${dt.dayName}</span>
                            <span class="text-lg font-extrabold">${dt.date}</span>
                            <span class="text-[10px] ${isFull?'text-harmony-error font-bold':'text-harmony-accent font-bold'}">${isFull?'PENUH':'Sisa: '+sisa}</span>
                        </button>`;
                    });
                }
                document.getElementById('day-buttons-container').innerHTML=h;
            });
        }
        function parseSchedule(str){ const s={}; str.split(',').forEach(d=>{ const m=d.trim().match(/^([a-zA-Z]+)\s*\((\d{2}:\d{2})\s*-\s*(\d{2}:\d{2})\)$/); if(m) s[m[1].toLowerCase()]={s:m[2],e:m[3]}; }); return s; }
        function genDates(sch){ const days=['minggu','senin','selasa','rabu','kamis','jumat','sabtu']; const r=[]; const t=new Date(); 
            for(let i=0; i<=14; i++){ const d=new Date(t); d.setDate(t.getDate()+i); const dn=days[d.getDay()]; if(sch[dn]) r.push({val:formatLocalDate(d), day:dn, dayName:dn.toUpperCase(), date:d.getDate()+'/'+(d.getMonth()+1)}); } return r; 
        }
        window.selDate=function(el, val, s, e){

            const res = document.getElementById('result-info');
            res.classList.remove('hidden');
            document.getElementById('result-date').textContent=val; document.getElementById('result-shift').textContent=s+' - '+e+' WIB';
            
            const btn = document.getElementById('btn-daftar');
            btn.disabled=false;
        }

        let isEditMode = false;
        const editableFields = ['prof-alamat', 'prof-telp', 'prof-email'];

        function toggleEditProfile() {
            if (isEditMode) {
                isEditMode = false;
                loadProfil(false); 
            } else {
                isEditMode = true;
            }
            
            const btn = document.getElementById('btn-toggle-edit');
            const saveContainer = document.getElementById('save-button-container');
            const saveBtn = document.getElementById('btn-simpan-profil');

            if (isEditMode) {
                btn.innerHTML = '<i class="fas fa-times-circle mr-1"></i> Batal Edit';
                btn.classList.remove('bg-harmony-primary', 'hover:bg-harmony-primary/90');
                btn.classList.add('bg-harmony-error', 'hover:bg-harmony-error/90');
                saveContainer.classList.remove('hidden');
                saveBtn.disabled = false;
                
                editableFields.forEach(id => {
                    const el = document.getElementById(id);
                    el.readOnly = false;
                    el.classList.remove('bg-slate-50');
                    el.classList.add('bg-white');
                });
            } else {
                btn.innerHTML = '<i class="fas fa-edit mr-1"></i> Edit Profil';
                btn.classList.remove('bg-harmony-error', 'hover:bg-harmony-error/90');
                btn.classList.add('bg-harmony-primary', 'hover:bg-harmony-primary/90');
                saveContainer.classList.add('hidden');
                saveBtn.disabled = true;
                
                editableFields.forEach(id => {
                    const el = document.getElementById(id);
                    el.readOnly = true;
                    el.classList.add('bg-slate-50');
                    el.classList.remove('bg-white');
                });
            }
        }

            e.preventDefault(); 
            const btn = e.submitter;
            
            // PERUBAHAN: Validasi minimal 8 karakter
            if (this.elements['new_password'].value.length < 8) {
                showToast('Password baru minimal 8 karakter.', 'error');

            btn.disabled = true;

            fetch(API_URL+'?action=forgot_password', {method:'POST', body:new FormData(this)}).then(r=>r.json()).then(d=>{
                showToast(d.message, d.status);
                btn.disabled = false;
                btn.innerHTML = 'Atur Ulang Password';

                if(d.status === 'success') {
                    closeModal('modal-forgot-password');
                    showLogin(); 
                }
            }).catch(e => {
                showToast("Gagal koneksi ke server.", "error");
                btn.disabled = false;
                btn.innerHTML = 'Atur Ulang Password';
            });
        });
